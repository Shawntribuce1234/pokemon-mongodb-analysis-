7)  What is the average HP for Pokemon grouped by habitat?


db.pokemon.aggregate([
  { $match: { habitat: { $ne: null } } },
  { $group: {
      _id: "$habitat",
      avg_hp: { $avg: "$stats.hp" },
      count:  { $sum: 1 }
  }},
  { $sort: { avg_hp: -1 } }
])

Output:
[
  { _id: 'rare', avg_hp: 95.2, count: 5 },
  { _id: 'mountain', avg_hp: 70.83333333333333, count: 18 },

9) What are the most common types  of Pokemon in my list?

db.pokemon.aggregate([
  { $unwind: "$types" },
  { $group: {
      _id:   "$types",
      count: { $sum: 1 }
  }},
  { $sort: { count: -1 } }
])

Output:
  { _id: 'poison', count: 33 },
  { _id: 'water', count: 32 },
  { _id: 'normal', count: 22 },

10) How do average attack and defense of legendary and non-legendary Pokemon compare to each other?

db.pokemon.aggregate([
  { $group: {
      _id:            "$is_legendary",
      avg_attack:     { $avg: "$stats.attack" },
      avg_defense:    { $avg: "$stats.defense" },
      avg_hp:         { $avg: "$stats.hp" },
      count:          { $sum: 1 }
  }},
  { $sort: { _id: -1 } }
])

Output:
  {
    _id: true,
    avg_attack: 96.25,
    avg_defense: 91.25,
